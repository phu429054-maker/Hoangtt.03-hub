local function new(class, props, parentObject)
    local object = Instance.new(class)
    for key, value in pairs(props) do
        object[key] = value
    end
    object.Parent = parentObject
    return object
end

local function animate(object, duration, props)
    local t = TweenService:Create(
        object,
        TweenInfo.new(duration, Enum.EasingStyle.Quint,
            Enum.EasingDirection.Out),
        props
    )
    t:Play()
    return t
end

local currentProgress = 0

local function setProgress(value, message)
    currentProgress = math.max(currentProgress, value)

    subtitle.Text = message:upper()
    percent.Text = string.format("%d%%", currentProgress)

    animate(fill, 0.22, {
        Size = UDim2.fromScale(currentProgress / 100, 1),
    })
end

local function closeLoader()
    if not gui.Parent then
        return
    end

    animate(card, 0.22, {
        Position = UDim2.fromScale(0.5, 0.46),
        BackgroundTransparency = 1,
    })

    animate(backdrop, 0.28, {
        BackgroundTransparency = 1,
    })

    task.wait(0.3)

    if gui.Parent then
        gui:Destroy()
    end
end

local function showError(message)
    if not gui.Parent then
        return
    end

    subtitle.Text = "DOWNLOAD FAILED"
    subtitle.TextColor3 = Color3.fromRGB(255, 85, 100)
    status.Text = tostring(message):sub(1, 180)
    status.TextColor3 = Color3.fromRGB(255, 125, 135)
    footer.Text = "CHECK CONNECTION OR SCRIPT URL"

    task.wait(2.5)
    closeLoader()
end

-- Staged visual progress while the HTTP request is in flight.
local stages = {
    {8,  "Connecting to server..."},
    {18, "Downloading hub data..."},
    {32, "Downloading image assets..."},
    {46, "Loading UI modules..."},
    {58, "Processing configuration..."},
    {68, "Preparing hub interface..."},
}

local finished = false

task.spawn(function()
    for _, stage in ipairs(stages) do
        if finished then
            return
        end

        setProgress(stage[1], stage[2])
        task.wait(0.12)
    end
end)

-- Fetch actual source; don't execute if the request fails.
local requestOK, source = pcall(function()
    return game:HttpGet(URL)
end)

finished = true

if not requestOK then
    showError("HTTP request failed: " .. tostring(source))
    return
end

if type(source) ~= "string" or source:match("^%s*$") then
    showError("Server returned an empty script.")
    return
end

if #source < 20 then
    showError("Downloaded response is too short.")
    return
end

setProgress(76, "Download complete - compiling...")

if type(loadstring) ~= "function" then
    showError("loadstring is unavailable in this environment.")
    return
end

local compileOK, chunk = pcall(loadstring, source)

if not compileOK or type(chunk) ~= "function" then
    showError("Compilation failed: " .. tostring(chunk))
    return
end

setProgress(88, "Initializing hub modules...")
task.wait(0.12)

setProgress(96, "Launching hub...")
task.wait(0.12)

setProgress(100, "Hub launched successfully!")

-- Run the hub separately because its main function may never return.
task.spawn(function()
    local runOK, runError = pcall(chunk)

    if not runOK then
        warn("[HOANGTT.03] Hub runtime error:", runError)
    end
end)

task.wait(0.25)
closeLoader()
